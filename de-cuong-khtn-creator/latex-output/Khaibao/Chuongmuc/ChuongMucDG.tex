% ===== Icon mặc định: nguyên tử TikZ vector, không dùng ảnh ngoài =====
\newcommand{\AtomIcon}{%
	\begin{tikzpicture}[baseline=-0.55ex,scale=0.42,line cap=round,line join=round]
		\shade[ball color=\mycolor!80] (0,0) circle (0.23);
		\foreach \ang in {0,60,-60}{%
			\begin{scope}[rotate=\ang]
				\draw[\maudam!80!teal,line width=1.2pt] (0,0) ellipse (1.05 and 0.33);
				\shade[ball color=\mycolor!80] (1.05,0) circle (0.11);
			\end{scope}}
	\end{tikzpicture}%
}
% ===== Key chỉnh chapterbox =====
% icon: biểu tượng bên trái nhãn; number: số chương; label: chữ nhãn chương.
% colback: nền thân hộp; colframe: viền; numberback: nền ô số.
% textcolor: màu chữ; arc: độ bo góc; fonttitle: font tên chương.
\newif\ifchapterboxdecoration
\pgfkeys{/chapterbox/.is family,/chapterbox,
	icon/.store in=\chapterboxicon,
    number/.store in=\chapterboxnumber,
    label/.store in=\chapterboxlabel,
	show-decoration/.is if=chapterboxdecoration,
	show-decoration/.default=true,
	icon=\AtomIcon,
	number=1,
	label=CHƯƠNG,
	show-decoration=true
}

\NewTColorBox{chapterbox}{ O{} }{%
	code={\pgfkeys{/chapterbox,#1}},
	enhanced,
	before skip=0mm,
	after skip=0mm,
	width=\linewidth,
	boxrule=0pt,
	frame hidden,
	arc=3mm,
	outer arc=3mm,
	colback=\mycolor!20,
	left=5mm,right=5mm,top=5mm,bottom=5mm,
	halign=center,
	drop fuzzy shadow={\mycolor!10!black},
	before upper={\fontsize{25pt}{10pt}\fontfamily{phv}\selectfont\color{\mycolor!30!black}\bfseries\centering\MakeUppercase},
	overlay={%
		\ifchapterboxdecoration
		\node[anchor=south east,inner sep=0pt] at ([xshift=-29mm,yshift=-1mm]frame.north east)
		{\fontfamily{phv}\bfseries\LARGE\color{\mycolor!30!black}\chapterboxicon\hspace{1.5mm}\chapterboxlabel};
		\draw[rounded corners=2mm,\mycolor!50!black,line width=1.25pt] ([shift={(-45:8pt)}]frame.north west)-- ([shift={(-135:8pt)}]frame.north east)--([shift={(135:8pt)}]frame.south east)--([shift={(45:8pt)}]frame.south west)--cycle;
		\node[anchor=center,rounded corners=1mm,draw=\mycolor!60!black,line width=1.35pt,
		fill=\mycolor!40,minimum width=17mm,minimum height=17mm,
		inner sep=1.5mm,text=\mycolor!20!black,font=\fontfamily{phv}\bfseries\bfseries\fontsize{34}{34}\selectfont]
		at ([xshift=-17mm,yshift=3.8mm]frame.north east) {\chapterboxnumber};
		\fi
	}
}
